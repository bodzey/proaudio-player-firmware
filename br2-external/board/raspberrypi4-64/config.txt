start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image
arm_64bit=1

# Audio appliance: never probe or expose a local video path.
disable_splash=1
boot_delay=0
disable_overscan=1
display_auto_detect=0
camera_auto_detect=0
max_framebuffers=0
enable_tvout=0

# VC4 KMS is retained only for HDMI audio. Keep CMA deliberately small because
# there is no framebuffer, camera, codec or 3D workload in ProAudio Player.
dtoverlay=vc4-kms-v3d,cma-64

# Expose the Raspberry Pi onboard analogue audio device to ALSA. The native
# player remains hardware-neutral and discovers it through PipeWire exactly like
# USB, HDMI and future I2S outputs.
dtparam=audio=on

# Bluetooth is not part of the product. Keep PL011 available for DEV serial SSH/
# console diagnostics; production disables UART separately.
dtoverlay=disable-bt
enable_uart=1
