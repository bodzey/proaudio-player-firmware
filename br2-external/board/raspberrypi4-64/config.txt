start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image
arm_64bit=1

# Headless audio appliance: no local UI, camera or composite-video path.
disable_splash=1
boot_delay=0
disable_overscan=1
display_auto_detect=0
camera_auto_detect=0
max_framebuffers=0
enable_tvout=0

# HDMI audio is provided by VC4/KMS. Linux owns EDID/mode discovery; no local
# framebuffer console, virtual terminal or V3D userspace stack is built.
dtoverlay=vc4-kms-v3d,cma-64
disable_fw_kms_setup=1

# Keep the Raspberry Pi 4 built-in 3.5 mm PWM output available in high-quality
# mode. Legacy snd_bcm2835 is restricted to headphones by cmdline.txt, while
# HDMI audio remains owned by VC4/KMS.
dtparam=audio=on
audio_pwm_mode=2

# Hand the BCM2835 hardware watchdog to Linux during boot. systemd services it
# continuously after PID 1 starts.
kernel_watchdog_timeout=30

# Bluetooth is not part of the product. PL011 remains available in the current
# DEV/recovery profile for serial diagnostics.
dtoverlay=disable-bt
enable_uart=1
